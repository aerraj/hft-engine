cmake_minimum_required(VERSION 3.24)
project(hft_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HFT_NATIVE "Native benchmark instructions" OFF)
set(HFT_SANITIZER "" CACHE STRING "address,undefined or thread")
add_library(hft INTERFACE)
target_include_directories(hft INTERFACE include)
target_compile_options(hft INTERFACE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
if(HFT_SANITIZER)
 target_compile_options(hft INTERFACE -fsanitize=${HFT_SANITIZER} -fno-omit-frame-pointer)
 target_link_options(hft INTERFACE -fsanitize=${HFT_SANITIZER})
endif()
find_package(Threads REQUIRED)
add_executable(hft_replay src/main.cpp)
target_link_libraries(hft_replay PRIVATE hft)
add_executable(hft_tests tests/test.cpp)
target_link_libraries(hft_tests PRIVATE hft Threads::Threads)
target_compile_options(hft_tests PRIVATE -UNDEBUG)
add_executable(hft_bench bench/latency.cpp)
target_link_libraries(hft_bench PRIVATE hft)
if(HFT_NATIVE)
 target_compile_options(hft_bench PRIVATE -O3 -march=native)
endif()
enable_testing()
add_test(NAME unit COMMAND hft_tests)
