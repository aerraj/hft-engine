cmake_minimum_required(VERSION 3.24)
project(hft_engine VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HFT_NATIVE "Native instructions in benchmark targets only" OFF)
option(HFT_TESTS "Build GoogleTest suite" ON)
option(HFT_GOOGLE_BENCHMARK "Build Google Benchmark microbenchmarks" ON)
option(HFT_FUZZ "Build Clang libFuzzer target" OFF)
set(HFT_SANITIZER "" CACHE STRING "address,undefined or thread")
add_library(hft INTERFACE)
target_include_directories(hft INTERFACE include)
target_compile_options(hft INTERFACE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
if(HFT_SANITIZER)
 add_compile_options(-fsanitize=${HFT_SANITIZER} -fno-omit-frame-pointer)
 add_link_options(-fsanitize=${HFT_SANITIZER})
endif()
find_package(Threads REQUIRED)
include(FetchContent)
set(HDR_HISTOGRAM_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(HDR_HISTOGRAM_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(HDR_LOG_REQUIRED DISABLED CACHE STRING "" FORCE)
FetchContent_Declare(hdr GIT_REPOSITORY https://github.com/HdrHistogram/HdrHistogram_c.git GIT_TAG 8dcce8f68512fca460b171bccc3a5afce0048779)
FetchContent_MakeAvailable(hdr)
target_link_libraries(hft INTERFACE hdr_histogram_static)
add_executable(hft_replay src/main.cpp)
target_link_libraries(hft_replay PRIVATE hft Threads::Threads)
add_executable(hft_live src/live.cpp)
target_link_libraries(hft_live PRIVATE hft Threads::Threads)
# Allocation audit overrides conflict with sanitizer runtime allocators.
# Performance binaries are deliberately separate from correctness builds.
if(NOT HFT_SANITIZER)
add_executable(hft_bench bench/latency.cpp)
target_link_libraries(hft_bench PRIVATE hft Threads::Threads)
target_compile_definitions(hft_bench PRIVATE HFT_BUILD_TYPE="${CMAKE_BUILD_TYPE}" HFT_NATIVE_ENABLED=$<BOOL:${HFT_NATIVE}>)
if(HFT_NATIVE)
 target_compile_options(hft_bench PRIVATE -O3 -march=native)
endif()
endif()
if(HFT_TESTS)
 enable_testing()
 find_package(GTest CONFIG QUIET)
 if(NOT GTest_FOUND)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG b514bdc898e2951020cbdca1304b75f5950d1f59)
  FetchContent_MakeAvailable(googletest)
 endif()
 add_executable(hft_tests tests/test.cpp)
 target_link_libraries(hft_tests PRIVATE hft GTest::gtest_main Threads::Threads)
 add_test(NAME unit_property COMMAND hft_tests)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME golden_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:hft_replay> ${CMAKE_CURRENT_SOURCE_DIR})
 add_test(NAME live_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/live_loopback.py $<TARGET_FILE:hft_live> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
if(HFT_GOOGLE_BENCHMARK)
 find_package(benchmark CONFIG QUIET)
 if(NOT benchmark_FOUND)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG 96afad55c79e02f5dfca1374e772c2be72ba631b)
  FetchContent_MakeAvailable(benchmark)
 endif()
 add_executable(hft_microbench bench/micro.cpp)
 target_link_libraries(hft_microbench PRIVATE hft benchmark::benchmark)
 if(HFT_NATIVE)
  target_compile_options(hft_microbench PRIVATE -O3 -march=native)
 endif()
endif()
if(HFT_FUZZ)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "libFuzzer requires Clang")
 endif()
 add_executable(hft_fuzz tests/fuzz.cpp)
 target_link_libraries(hft_fuzz PRIVATE hft)
 target_compile_options(hft_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
 target_link_options(hft_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
